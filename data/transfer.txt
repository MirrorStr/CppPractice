struct DevBasicConfig : public DevData_t  // 设备基本参数配置
{
    string m_strDevNmae;      /* 设备名称 */
    string m_strLanguage;     /* 设备语言 */
    string m_strCountryCode;  // 国家码
    DevBasicConfig()
        : m_strDevNmae("Device")
        , m_strLanguage("chinese")
        , m_strCountryCode("")
    {
    }
    bool operator==(const DevBasicConfig& t) const
    {
        return ((this->m_strDevNmae == t.m_strDevNmae) && (this->m_strLanguage == t.m_strLanguage)
                && (this->m_strCountryCode == t.m_strCountryCode));
    }
    DevBasicConfig& operator=(const DevBasicConfig& t)
    {
        this->m_strDevNmae     = t.m_strDevNmae;
        this->m_strLanguage    = t.m_strLanguage;
        this->m_strCountryCode = t.m_strCountryCode;
        return *this;
    }
};

struct Wire4Config  //4线参数配置
{

    int8_t m_iOutputGain;  //输出增益

    Wire4Config& operator=(const Wire4Config& t)
    {
        this->m_iLineNum    = t.m_iLineNum;
        this->m_iInputGain  = t.m_iInputGain;
        this->m_iOutputGain = t.m_iOutputGain;
        this->m_iLineSeq    = t.m_iLineSeq;
        return *this;
    }
};

    }

                && (this->m_iOutputGain == t.m_iOutputGain) && (this->m_iLineSeq == t.m_iLineSeq)
                && (this->m_strCable == t.m_strCable) && (this->m_bRessitance == t.m_bRessitance)
                && (this->m_bZero == t.m_bZero) && (this->m_strLineMode == t.m_strLineMode));
    }
    Wire2Config& operator=(const Wire2Config& t)
    {
        this->m_iLineNum    = t.m_iLineNum;
        this->m_iInputGain  = t.m_iInputGain;
        this->m_iOutputGain = t.m_iOutputGain;
        this->m_iLineSeq    = t.m_iLineSeq;
        this->m_strCable    = t.m_strCable;
        this->m_bRessitance = t.m_bRessitance;
        this->m_bZero       = t.m_bZero;
        this->m_strLineMode = t.m_strLineMode;
        return *this;
    }
};

struct DevKey /* MS的面板按键设置 仅MS存在 */
